using System;
using System.Collections.Generic;
using UnityEngine;
using Herghys.MultipleAdsManager.Runtime;

#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
using UnityEngine.Advertisements;
using RuntimeBannerPosition = Herghys.MultipleAdsManager.Runtime.BannerPosition;
using SdkBannerPosition = UnityEngine.Advertisements.BannerPosition;
#else
using RuntimeBannerPosition = Herghys.MultipleAdsManager.Runtime.BannerPosition;
#endif

namespace Herghys.MultipleAdsManager.Modules
{
    public class UnityAdsProvider : IAdProvider
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
        , IUnityAdsInitializationListener, IUnityAdsLoadListener, IUnityAdsShowListener
#endif
    {
        public string NetworkName => "UnityAds";
        public AdNetworkType NetworkType => AdNetworkType.UnityAds;
        public bool IsInitialized => isInitialized;

        public int BannerWeight { get; set; } = 50;
        public int InterstitialWeight { get; set; } = 50;
        public int RewardedWeight { get; set; } = 50;

        public bool IsBannerSupported => true;
        public bool IsInterstitialReady => isInterstitialReady;
        public bool IsRewardedReady => isRewardedReady;

#pragma warning disable 0067
        public event Action<IAdsType> OnAdLoaded;
        public event Action<IAdsType, string> OnAdFailedToLoad;
        public event Action<IAdsType> OnAdShown;
        public event Action<IAdsType> OnAdClosed;
        public event Action OnRewardEarned;
        public event Action<AdReward> OnRewardEarnedWithReward;
#pragma warning restore 0067

        private BaseAdConfig config;
        private bool isInitialized = false;
        private bool isInterstitialReady = false;
        private bool isRewardedReady = false;

#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
        private Action<bool> onInitializationCallback;
        private Action onCurrentInterstitialClosed;
        private Action onCurrentRewardedClosed;
        private Action<AdReward> onCurrentRewardEarned;

        private string lastBannerAdUnitId;
        private string lastInterstitialAdUnitId;
        private string lastInterstitialPlacement;
        private string lastRewardedAdUnitId;
        private string lastRewardedPlacement;
#endif

        public int GetWeight(IAdsType adType)
        {
            switch (adType)
            {
                case IAdsType.Banner: return BannerWeight;
                case IAdsType.Interstitial: return InterstitialWeight;
                case IAdsType.Rewarded: return RewardedWeight;
                default: return 50;
            }
        }

        public void SetWeight(IAdsType adType, int weight)
        {
            switch (adType)
            {
                case IAdsType.Banner: BannerWeight = weight; break;
                case IAdsType.Interstitial: InterstitialWeight = weight; break;
                case IAdsType.Rewarded: RewardedWeight = weight; break;
            }
        }

        public int GetWeight(AdType adType) => GetWeight((IAdsType)adType);
        public void SetWeight(AdType adType, int weight) => SetWeight((IAdsType)adType, weight);

        public void Initialize(BaseAdConfig sourceConfig, List<string> testDeviceIds, bool targetAdsForChildren = false, Action<bool> onInitialized = null)
        {
            config = sourceConfig;
            if (config != null)
            {
                BannerWeight = config.BannerWeight;
                InterstitialWeight = config.InterstitialWeight;
                RewardedWeight = config.RewardedWeight;
            }

#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            if (targetAdsForChildren)
            {
                MetaData metaData = new MetaData("user");
                metaData.Set("nonbehavioral", "true");
                Advertisement.SetMetaData(metaData);
                Debug.Log("[UnityAdsProvider] Set nonbehavioral=true for COPPA compliance.");
            }

            if (Advertisement.isInitialized)
            {
                isInitialized = true;
                onInitialized?.Invoke(true);
                return;
            }

            string gameId = config?.AppId ?? "";
            if (string.IsNullOrEmpty(gameId))
            {
                Debug.LogWarning("[UnityAdsProvider] Game ID is empty. Running in stub mode until Game ID is configured.");
                isInitialized = true;
                onInitialized?.Invoke(true);
                return;
            }

            onInitializationCallback = onInitialized;
            bool testMode = config != null && config.UseTestAdIds;
            Advertisement.Initialize(gameId, testMode, this);
#elif UNITY_ADS
            Debug.LogWarning("[UnityAdsProvider] UNITY_ADS is enabled, but Unity Ads SDK (com.unity.ads) is not yet detected/installed. Running in stub mode. Config remains active.");
            isInitialized = true;
            onInitialized?.Invoke(true);
#else
            Debug.LogWarning("[UnityAdsProvider] UNITY_ADS scripting symbol is not defined. Unity Ads is running in stub mode.");
            isInitialized = true;
            onInitialized?.Invoke(true);
#endif
        }

#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
        public void OnInitializationComplete()
        {
            isInitialized = true;
            Debug.Log("[UnityAdsProvider] Unity Ads SDK initialized successfully.");
            onInitializationCallback?.Invoke(true);
            onInitializationCallback = null;
        }

        public void OnInitializationFailed(UnityAdsInitializationError error, string message)
        {
            Debug.LogError($"[UnityAdsProvider] Unity Ads initialization failed: {error} - {message}");
            onInitializationCallback?.Invoke(false);
            onInitializationCallback = null;
        }

        private SdkBannerPosition ConvertBannerPosition(RuntimeBannerPosition position)
        {
            switch (position)
            {
                case RuntimeBannerPosition.Top: return SdkBannerPosition.TOP_CENTER;
                case RuntimeBannerPosition.TopLeft: return SdkBannerPosition.TOP_LEFT;
                case RuntimeBannerPosition.TopRight: return SdkBannerPosition.TOP_RIGHT;
                case RuntimeBannerPosition.BottomLeft: return SdkBannerPosition.BOTTOM_LEFT;
                case RuntimeBannerPosition.BottomRight: return SdkBannerPosition.BOTTOM_RIGHT;
                case RuntimeBannerPosition.Center: return SdkBannerPosition.CENTER;
                case RuntimeBannerPosition.Bottom:
                default:
                    return SdkBannerPosition.BOTTOM_CENTER;
            }
        }
#endif

        public void ShowBanner(RuntimeBannerPosition position, string adUnitId = null, string placement = null)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            if (string.IsNullOrEmpty(adUnitId))
            {
                Debug.LogWarning($"[UnityAdsProvider] Banner ad unit ID for {position} is empty.");
                return;
            }

            lastBannerAdUnitId = adUnitId;
            Advertisement.Banner.SetPosition(ConvertBannerPosition(position));

            var loadOptions = new BannerLoadOptions
            {
                loadCallback = () =>
                {
                    OnAdLoaded?.Invoke(IAdsType.Banner);
                    var showOptions = new BannerOptions
                    {
                        showCallback = () => OnAdShown?.Invoke(IAdsType.Banner),
                        hideCallback = () => OnAdClosed?.Invoke(IAdsType.Banner)
                    };
                    Advertisement.Banner.Show(adUnitId, showOptions);
                },
                errorCallback = message =>
                {
                    Debug.LogWarning($"[UnityAdsProvider] Banner load failed: {message}");
                    OnAdFailedToLoad?.Invoke(IAdsType.Banner, message);
                }
            };

            Advertisement.Banner.Load(adUnitId, loadOptions);
#else
            Debug.Log($"[UnityAdsProvider-Stub] ShowBanner ({position}, ID: {adUnitId}, Placement: {placement})");
            OnAdShown?.Invoke(IAdsType.Banner);
#endif
        }

        public void HideBanner(RuntimeBannerPosition position)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            Advertisement.Banner.Hide(false);
#else
            Debug.Log($"[UnityAdsProvider-Stub] HideBanner ({position})");
#endif
        }

        public void DestroyBanner(RuntimeBannerPosition position)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            Advertisement.Banner.Hide(true);
#else
            Debug.Log($"[UnityAdsProvider-Stub] DestroyBanner ({position})");
#endif
        }

        public void LoadInterstitial(string adUnitId = null, string placement = null)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            if (string.IsNullOrEmpty(adUnitId)) return;
            lastInterstitialAdUnitId = adUnitId;
            lastInterstitialPlacement = placement;
            isInterstitialReady = false;
            Advertisement.Load(adUnitId, this);
#else
            Debug.Log($"[UnityAdsProvider-Stub] Loading Interstitial: {adUnitId}");
            isInterstitialReady = true;
            OnAdLoaded?.Invoke(IAdsType.Interstitial);
#endif
        }

        public void ShowInterstitial(Action onClosed = null)
        {
            ShowInterstitial(null, null, onClosed);
        }

        public void ShowInterstitial(string adUnitId, string placement, Action onClosed = null)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            string unitToShow = !string.IsNullOrEmpty(adUnitId) ? adUnitId : lastInterstitialAdUnitId;
            if (IsInterstitialReady && !string.IsNullOrEmpty(unitToShow))
            {
                onCurrentInterstitialClosed = onClosed;
                isInterstitialReady = false;
                Advertisement.Show(unitToShow, this);
            }
            else
            {
                onClosed?.Invoke();
                if (!string.IsNullOrEmpty(unitToShow)) LoadInterstitial(unitToShow, placement);
            }
#else
            Debug.Log($"[UnityAdsProvider-Stub] Showing Interstitial: {adUnitId} (Placement: {placement})");
            OnAdShown?.Invoke(IAdsType.Interstitial);
            OnAdClosed?.Invoke(IAdsType.Interstitial);
            onClosed?.Invoke();
#endif
        }

        public void LoadRewarded(string adUnitId = null, string placement = null)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            if (string.IsNullOrEmpty(adUnitId)) return;
            lastRewardedAdUnitId = adUnitId;
            lastRewardedPlacement = placement;
            isRewardedReady = false;
            Advertisement.Load(adUnitId, this);
#else
            Debug.Log($"[UnityAdsProvider-Stub] Loading Rewarded: {adUnitId}");
            isRewardedReady = true;
            OnAdLoaded?.Invoke(IAdsType.Rewarded);
#endif
        }

        public void ShowRewarded(Action onRewardEarned, Action onClosed = null)
        {
            ShowRewarded(null, null, onRewardEarned, onClosed);
        }

        public void ShowRewarded(string adUnitId, string placement, Action onRewardEarned, Action onClosed = null)
        {
            ShowRewarded(adUnitId, placement, (reward) => onRewardEarned?.Invoke(), onClosed);
        }

        public void ShowRewarded(string adUnitId, string placement, Action<AdReward> onRewardEarned, Action onClosed = null)
        {
#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
            string unitToShow = !string.IsNullOrEmpty(adUnitId) ? adUnitId : lastRewardedAdUnitId;
            if (IsRewardedReady && !string.IsNullOrEmpty(unitToShow))
            {
                onCurrentRewardedClosed = onClosed;
                onCurrentRewardEarned = onRewardEarned;
                isRewardedReady = false;
                Advertisement.Show(unitToShow, this);
            }
            else
            {
                onClosed?.Invoke();
                if (!string.IsNullOrEmpty(unitToShow)) LoadRewarded(unitToShow, placement);
            }
#else
            var reward = new AdReward("Reward", 1);
            Debug.Log($"[UnityAdsProvider-Stub] Showing Rewarded: {adUnitId} (Placement: {placement})");
            OnAdShown?.Invoke(IAdsType.Rewarded);
            OnRewardEarned?.Invoke();
            OnRewardEarnedWithReward?.Invoke(reward);
            onRewardEarned?.Invoke(reward);
            OnAdClosed?.Invoke(IAdsType.Rewarded);
            onClosed?.Invoke();
#endif
        }

#if UNITY_ADS && UNITY_ADS_SDK_INSTALLED
        public void OnUnityAdsAdLoaded(string placementId)
        {
            if (!string.IsNullOrEmpty(lastInterstitialAdUnitId) && string.Equals(placementId, lastInterstitialAdUnitId, StringComparison.Ordinal))
            {
                isInterstitialReady = true;
                Debug.Log($"[UnityAdsProvider] Interstitial loaded: {placementId}");
                OnAdLoaded?.Invoke(IAdsType.Interstitial);
            }
            else if (!string.IsNullOrEmpty(lastRewardedAdUnitId) && string.Equals(placementId, lastRewardedAdUnitId, StringComparison.Ordinal))
            {
                isRewardedReady = true;
                Debug.Log($"[UnityAdsProvider] Rewarded loaded: {placementId}");
                OnAdLoaded?.Invoke(IAdsType.Rewarded);
            }
        }

        public void OnUnityAdsFailedToLoad(string placementId, UnityAdsLoadError error, string message)
        {
            Debug.LogWarning($"[UnityAdsProvider] Failed to load '{placementId}': {error} - {message}");
            if (!string.IsNullOrEmpty(lastInterstitialAdUnitId) && string.Equals(placementId, lastInterstitialAdUnitId, StringComparison.Ordinal))
            {
                isInterstitialReady = false;
                OnAdFailedToLoad?.Invoke(IAdsType.Interstitial, message);
            }
            else if (!string.IsNullOrEmpty(lastRewardedAdUnitId) && string.Equals(placementId, lastRewardedAdUnitId, StringComparison.Ordinal))
            {
                isRewardedReady = false;
                OnAdFailedToLoad?.Invoke(IAdsType.Rewarded, message);
            }
        }

        public void OnUnityAdsShowStart(string placementId)
        {
            if (!string.IsNullOrEmpty(lastInterstitialAdUnitId) && string.Equals(placementId, lastInterstitialAdUnitId, StringComparison.Ordinal))
            {
                OnAdShown?.Invoke(IAdsType.Interstitial);
            }
            else if (!string.IsNullOrEmpty(lastRewardedAdUnitId) && string.Equals(placementId, lastRewardedAdUnitId, StringComparison.Ordinal))
            {
                OnAdShown?.Invoke(IAdsType.Rewarded);
            }
        }

        public void OnUnityAdsShowClick(string placementId) { }

        public void OnUnityAdsShowFailure(string placementId, UnityAdsShowError error, string message)
        {
            Debug.LogWarning($"[UnityAdsProvider] Failed to show '{placementId}': {error} - {message}");
            if (!string.IsNullOrEmpty(lastInterstitialAdUnitId) && string.Equals(placementId, lastInterstitialAdUnitId, StringComparison.Ordinal))
            {
                onCurrentInterstitialClosed?.Invoke();
                onCurrentInterstitialClosed = null;
            }
            else if (!string.IsNullOrEmpty(lastRewardedAdUnitId) && string.Equals(placementId, lastRewardedAdUnitId, StringComparison.Ordinal))
            {
                onCurrentRewardedClosed?.Invoke();
                onCurrentRewardedClosed = null;
                onCurrentRewardEarned = null;
            }
        }

        public void OnUnityAdsShowComplete(string placementId, UnityAdsShowCompletionState showCompletionState)
        {
            if (!string.IsNullOrEmpty(lastInterstitialAdUnitId) && string.Equals(placementId, lastInterstitialAdUnitId, StringComparison.Ordinal))
            {
                OnAdClosed?.Invoke(IAdsType.Interstitial);
                onCurrentInterstitialClosed?.Invoke();
                onCurrentInterstitialClosed = null;
                LoadInterstitial(lastInterstitialAdUnitId, lastInterstitialPlacement);
            }
            else if (!string.IsNullOrEmpty(lastRewardedAdUnitId) && string.Equals(placementId, lastRewardedAdUnitId, StringComparison.Ordinal))
            {
                if (showCompletionState == UnityAdsShowCompletionState.COMPLETED)
                {
                    var adReward = new AdReward("Reward", 1);
                    OnRewardEarned?.Invoke();
                    OnRewardEarnedWithReward?.Invoke(adReward);
                    onCurrentRewardEarned?.Invoke(adReward);
                }

                OnAdClosed?.Invoke(IAdsType.Rewarded);
                onCurrentRewardedClosed?.Invoke();
                onCurrentRewardedClosed = null;
                onCurrentRewardEarned = null;
                LoadRewarded(lastRewardedAdUnitId, lastRewardedPlacement);
            }
        }
#endif
    }
}
