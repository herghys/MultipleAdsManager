using System;
using System.Collections.Generic;
using UnityEngine;
using Herghys.MultipleAdsManager.Runtime;

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
using GoogleMobileAds.Api;
#endif

namespace Herghys.MultipleAdsManager.Modules
{
    public class GoogleAdsProvider : IAdProvider
    {
        public string NetworkName => "GoogleMobileAds";
        public AdNetworkType NetworkType => AdNetworkType.GoogleAds;
        public bool IsInitialized => isInitialized;

        public int BannerWeight { get; set; } = 50;
        public int InterstitialWeight { get; set; } = 50;
        public int RewardedWeight { get; set; } = 50;

        public bool IsBannerSupported => true;

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
        public bool IsInterstitialReady => interstitialAd != null && interstitialAd.CanShowAd();
        public bool IsRewardedReady => rewardedAd != null && rewardedAd.CanShowAd();
#else
        public bool IsInterstitialReady => false;
        public bool IsRewardedReady => false;
#endif

        public event Action<IAdsType> OnAdLoaded;
        public event Action<IAdsType, string> OnAdFailedToLoad;
        public event Action<IAdsType> OnAdShown;
        public event Action<IAdsType> OnAdClosed;
        public event Action OnRewardEarned;
        public event Action<AdReward> OnRewardEarnedWithReward;

        private bool isInitialized = false;

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
        private readonly Dictionary<BannerPosition, BannerView> bannerViews = new Dictionary<BannerPosition, BannerView>();
        private InterstitialAd interstitialAd;
        private RewardedAd rewardedAd;

        private Action onCurrentInterstitialOpened;
        private Action onCurrentInterstitialClosed;
        private Action onCurrentRewardedOpened;
        private Action onCurrentRewardedClosed;
        private Action<AdReward> onCurrentRewardEarned;

        private string lastInterstitialAdUnitId;
        private string lastInterstitialPlacement;
        private string lastRewardedAdUnitId;
        private string lastRewardedPlacement;
#endif

        public int GetWeight(IAdsType adType)
        {
            switch (adType)
            {
                case IAdsType.Banner: return BannerWeight;
                case IAdsType.Interstitial: return InterstitialWeight;
                case IAdsType.Rewarded: return RewardedWeight;
                default: return 50;
            }
        }

        public void SetWeight(IAdsType adType, int weight)
        {
            switch (adType)
            {
                case IAdsType.Banner: BannerWeight = weight; break;
                case IAdsType.Interstitial: InterstitialWeight = weight; break;
                case IAdsType.Rewarded: RewardedWeight = weight; break;
            }
        }

        public int GetWeight(AdType adType) => GetWeight((IAdsType)adType);
        public void SetWeight(AdType adType, int weight) => SetWeight((IAdsType)adType, weight);

        public void Initialize(BaseAdConfig config, List<string> testDeviceIds, bool targetAdsForChildren = false, Action<bool> onInitialized = null)
        {
            if (config != null)
            {
                BannerWeight = config.BannerWeight;
                InterstitialWeight = config.InterstitialWeight;
                RewardedWeight = config.RewardedWeight;
            }

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            var reqConfig = new RequestConfiguration();

            if (testDeviceIds != null && testDeviceIds.Count > 0)
            {
                reqConfig.TestDeviceIds = testDeviceIds;
            }

            if (targetAdsForChildren)
            {
                reqConfig.AgeRestrictedTreatment = AgeRestrictedTreatment.Child;
                reqConfig.MaxAdContentRating = MaxAdContentRating.G;
            }

            MobileAds.SetRequestConfiguration(reqConfig);

            MobileAds.Initialize(initStatus =>
            {
                isInitialized = true;
                Debug.Log("[GoogleAdsProvider] Initialized successfully with Google Mobile Ads SDK.");
                onInitialized?.Invoke(true);
            });
#elif GOOGLE_ADS
            Debug.LogWarning("[GoogleAdsProvider] GOOGLE_ADS is enabled, but the Google Mobile Ads SDK is not yet detected/installed. Running in stub mode. Config remains active.");
            isInitialized = true;
            onInitialized?.Invoke(true);
#else
            Debug.LogWarning("[GoogleAdsProvider] GOOGLE_ADS scripting symbol is not defined. Google Mobile Ads is running in stub mode.");
            isInitialized = true;
            onInitialized?.Invoke(true);
#endif
        }

        #region Banner (All Positions)

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
        private AdPosition ConvertPosition(BannerPosition position)
        {
            switch (position)
            {
                case BannerPosition.Top: return AdPosition.Top;
                case BannerPosition.Bottom: return AdPosition.Bottom;
                case BannerPosition.TopLeft: return AdPosition.TopLeft;
                case BannerPosition.TopRight: return AdPosition.TopRight;
                case BannerPosition.BottomLeft: return AdPosition.BottomLeft;
                case BannerPosition.BottomRight: return AdPosition.BottomRight;
                case BannerPosition.Center: return AdPosition.Center;
                default: return AdPosition.Bottom;
            }
        }
#endif

        public void ShowBanner(BannerPosition position, string adUnitId = null, string placement = null)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (string.IsNullOrEmpty(adUnitId))
            {
                Debug.LogWarning($"[GoogleAdsProvider] Banner ad unit ID for {position} is empty.");
                return;
            }

            if (bannerViews.TryGetValue(position, out var existingBanner) && existingBanner != null)
            {
                existingBanner.Show();
                return;
            }

            AdPosition googlePos = ConvertPosition(position);
            var bannerView = new BannerView(adUnitId, AdSize.Banner, googlePos);
            AttachBannerEvents(bannerView, position);
            bannerViews[position] = bannerView;
            bannerView.LoadAd(new AdRequest());
            bannerView.Show();
#else
            Debug.Log($"[GoogleAdsProvider-Stub] ShowBanner ({position}, ID: {adUnitId}, Placement: {placement})");
#endif
        }

        public void HideBanner(BannerPosition position)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (bannerViews.TryGetValue(position, out var bannerView) && bannerView != null)
            {
                bannerView.Hide();
            }
#endif
        }

        public void DestroyBanner(BannerPosition position)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (bannerViews.TryGetValue(position, out var bannerView) && bannerView != null)
            {
                bannerView.Hide();
                bannerView.Destroy();
                bannerViews.Remove(position);
            }
#endif
        }

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
        private void AttachBannerEvents(BannerView banner, BannerPosition pos)
        {
            banner.OnBannerAdLoaded += () =>
            {
                Debug.Log($"[GoogleAdsProvider] {pos} Banner Loaded.");
                OnAdLoaded?.Invoke(IAdsType.Banner);
            };

            banner.OnBannerAdLoadFailed += (LoadAdError error) =>
            {
                Debug.LogWarning($"[GoogleAdsProvider] {pos} Banner Load Failed: {error.GetMessage()}");
                OnAdFailedToLoad?.Invoke(IAdsType.Banner, error.GetMessage());
            };
        }
#endif

        #endregion

        #region Interstitial

        public void LoadInterstitial(string adUnitId = null, string placement = null)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (string.IsNullOrEmpty(adUnitId))
            {
                Debug.LogWarning("[GoogleAdsProvider] Interstitial ad unit ID is empty.");
                return;
            }

            lastInterstitialAdUnitId = adUnitId;
            lastInterstitialPlacement = placement;

            if (interstitialAd != null)
            {
                interstitialAd.Destroy();
                interstitialAd = null;
            }

            InterstitialAd.Load(adUnitId, new AdRequest(), (InterstitialAd ad, LoadAdError error) =>
            {
                if (error != null || ad == null)
                {
                    Debug.LogWarning($"[GoogleAdsProvider] Interstitial failed to load: {error?.GetMessage()}");
                    OnAdFailedToLoad?.Invoke(IAdsType.Interstitial, error?.GetMessage());
                    return;
                }

                interstitialAd = ad;
                AttachInterstitialEvents(interstitialAd);
                Debug.Log("[GoogleAdsProvider] Interstitial loaded successfully.");
                OnAdLoaded?.Invoke(IAdsType.Interstitial);
            });
#endif
        }

        public void ShowInterstitial(Action onClosed = null)
        {
            ShowInterstitial(null, null, null, onClosed);
        }

        public void ShowInterstitial(Action onOpened, Action onClosed)
        {
            ShowInterstitial(null, null, onOpened, onClosed);
        }

        public void ShowInterstitial(string adUnitId, string placement, Action onClosed = null)
        {
            ShowInterstitial(adUnitId, placement, null, onClosed);
        }

        public void ShowInterstitial(string adUnitId, string placement, Action onOpened, Action onClosed)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (IsInterstitialReady)
            {
                onCurrentInterstitialOpened = onOpened;
                onCurrentInterstitialClosed = onClosed;
                interstitialAd.Show();
            }
            else
            {
                Debug.LogWarning("[GoogleAdsProvider] Interstitial is not ready. Loading new one...");
                onClosed?.Invoke();
                if (!string.IsNullOrEmpty(adUnitId)) LoadInterstitial(adUnitId, placement);
            }
#else
            Debug.Log($"[GoogleAdsProvider-Stub] ShowInterstitial (ID: {adUnitId}, Placement: {placement})");
            onOpened?.Invoke();
            onClosed?.Invoke();
#endif
        }

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
        private void AttachInterstitialEvents(InterstitialAd ad)
        {
            ad.OnAdFullScreenContentOpened += () =>
            {
                Debug.Log("[GoogleAdsProvider] Interstitial Opened.");
                OnAdShown?.Invoke(IAdsType.Interstitial);
                onCurrentInterstitialOpened?.Invoke();
                onCurrentInterstitialOpened = null;
            };

            ad.OnAdFullScreenContentClosed += () =>
            {
                Debug.Log("[GoogleAdsProvider] Interstitial Closed.");
                OnAdClosed?.Invoke(IAdsType.Interstitial);
                onCurrentInterstitialClosed?.Invoke();
                onCurrentInterstitialClosed = null;

                if (!string.IsNullOrEmpty(lastInterstitialAdUnitId))
                {
                    LoadInterstitial(lastInterstitialAdUnitId, lastInterstitialPlacement);
                }
            };

            ad.OnAdFullScreenContentFailed += (AdError error) =>
            {
                Debug.LogWarning($"[GoogleAdsProvider] Interstitial Failed to Show: {error.GetMessage()}");
                onCurrentInterstitialOpened = null;
                onCurrentInterstitialClosed?.Invoke();
                onCurrentInterstitialClosed = null;
            };
        }
#endif

        #endregion

        #region Rewarded

        public void LoadRewarded(string adUnitId = null, string placement = null)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (string.IsNullOrEmpty(adUnitId))
            {
                Debug.LogWarning("[GoogleAdsProvider] Rewarded ad unit ID is empty.");
                return;
            }

            lastRewardedAdUnitId = adUnitId;
            lastRewardedPlacement = placement;

            if (rewardedAd != null)
            {
                rewardedAd.Destroy();
                rewardedAd = null;
            }

            RewardedAd.Load(adUnitId, new AdRequest(), (RewardedAd ad, LoadAdError error) =>
            {
                if (error != null || ad == null)
                {
                    Debug.LogWarning($"[GoogleAdsProvider] Rewarded ad failed to load: {error?.GetMessage()}");
                    OnAdFailedToLoad?.Invoke(IAdsType.Rewarded, error?.GetMessage());
                    return;
                }

                rewardedAd = ad;
                AttachRewardedEvents(rewardedAd);
                Debug.Log("[GoogleAdsProvider] Rewarded ad loaded successfully.");
                OnAdLoaded?.Invoke(IAdsType.Rewarded);
            });
#endif
        }

        public void ShowRewarded(Action onRewardEarned, Action onClosed = null)
        {
            ShowRewarded(null, null, (reward) => onRewardEarned?.Invoke(), null, onClosed);
        }

        public void ShowRewarded(Action onRewardEarned, Action onOpened, Action onClosed)
        {
            ShowRewarded(null, null, (reward) => onRewardEarned?.Invoke(), onOpened, onClosed);
        }

        public void ShowRewarded(string adUnitId, string placement, Action onRewardEarned, Action onClosed = null)
        {
            ShowRewarded(adUnitId, placement, (reward) => onRewardEarned?.Invoke(), null, onClosed);
        }

        public void ShowRewarded(string adUnitId, string placement, Action onRewardEarned, Action onOpened, Action onClosed)
        {
            ShowRewarded(adUnitId, placement, (reward) => onRewardEarned?.Invoke(), onOpened, onClosed);
        }

        public void ShowRewarded(string adUnitId, string placement, Action<AdReward> onRewardEarned, Action onClosed = null)
        {
            ShowRewarded(adUnitId, placement, onRewardEarned, null, onClosed);
        }

        public void ShowRewarded(string adUnitId, string placement, Action<AdReward> onRewardEarned, Action onOpened, Action onClosed)
        {
#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
            if (IsRewardedReady)
            {
                onCurrentRewardedOpened = onOpened;
                onCurrentRewardedClosed = onClosed;
                onCurrentRewardEarned = onRewardEarned;

                rewardedAd.Show(reward =>
                {
                    var adReward = new AdReward(reward.Type, reward.Amount);
                    Debug.Log($"[GoogleAdsProvider] User earned reward: {adReward.ItemName} x {adReward.Amount}");
                    OnRewardEarned?.Invoke();
                    OnRewardEarnedWithReward?.Invoke(adReward);
                    onCurrentRewardEarned?.Invoke(adReward);
                });
            }
            else
            {
                Debug.LogWarning("[GoogleAdsProvider] Rewarded ad is not ready. Loading new one...");
                onClosed?.Invoke();
                if (!string.IsNullOrEmpty(adUnitId)) LoadRewarded(adUnitId, placement);
            }
#else
            Debug.Log($"[GoogleAdsProvider-Stub] ShowRewarded (ID: {adUnitId}, Placement: {placement})");
            onOpened?.Invoke();
            onRewardEarned?.Invoke(new AdReward("StubReward", 1));
            onClosed?.Invoke();
#endif
        }

#if GOOGLE_ADS && GOOGLE_ADS_SDK_INSTALLED
        private void AttachRewardedEvents(RewardedAd ad)
        {
            ad.OnAdFullScreenContentOpened += () =>
            {
                Debug.Log("[GoogleAdsProvider] Rewarded ad opened.");
                OnAdShown?.Invoke(IAdsType.Rewarded);
                onCurrentRewardedOpened?.Invoke();
                onCurrentRewardedOpened = null;
            };

            ad.OnAdFullScreenContentClosed += () =>
            {
                Debug.Log("[GoogleAdsProvider] Rewarded ad closed.");
                OnAdClosed?.Invoke(IAdsType.Rewarded);
                onCurrentRewardedClosed?.Invoke();
                onCurrentRewardedClosed = null;

                if (!string.IsNullOrEmpty(lastRewardedAdUnitId))
                {
                    LoadRewarded(lastRewardedAdUnitId, lastRewardedPlacement);
                }
            };

            ad.OnAdFullScreenContentFailed += (AdError error) =>
            {
                Debug.LogWarning($"[GoogleAdsProvider] Rewarded ad failed to show: {error.GetMessage()}");
                onCurrentRewardedOpened = null;
                onCurrentRewardedClosed?.Invoke();
                onCurrentRewardedClosed = null;
            };
        }
#endif

        #endregion
    }
}
